İlk kurulum (bir kez yapılır)

Supabase → SQL Editor bölümünde aşağıdaki SQL'i çalıştır. Sonra iki oyuncu da aynı oyunda çevrim içi hesaba giriş yapmalı.

create table if not exists public.c48_rps_rooms (id uuid primary key default gen_random_uuid(), code text unique not null, host_id uuid not null references auth.users(id) on delete cascade, guest_id uuid references auth.users(id) on delete set null, host_name text not null, guest_name text, bet integer not null check (bet between 1 and 1000), host_choice text check (host_choice in ('rock','paper','scissors')), guest_choice text check (guest_choice in ('rock','paper','scissors')), status text not null default 'waiting' check (status in ('waiting','playing','done')), winner_id uuid, created_at timestamptz not null default now());
alter table public.c48_rps_rooms enable row level security;
create policy "players read own rooms" on public.c48_rps_rooms for select to authenticated using (auth.uid()=host_id or auth.uid()=guest_id or status='waiting');
create policy "players create rooms" on public.c48_rps_rooms for insert to authenticated with check (auth.uid()=host_id);
create policy "players update joined rooms" on public.c48_rps_rooms for update to authenticated using (auth.uid()=host_id or auth.uid()=guest_id or (status='waiting' and guest_id is null)) with check (auth.uid()=host_id or auth.uid()=guest_id);
